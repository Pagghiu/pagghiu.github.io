101inline ResultErrorFormat formatFileErrorWithContext(FileError error, FileErrorDetail detail,
102 FileErrorContextKind contextKind, FileErrorContext context,
103 Span<char> output)
104{
105 ResultErrorFormatter formatter(output);
106 if (not appendFileError(formatter, error))
107 return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
108 if (detail != FileErrorDetail::None)
109 {
110 formatter.append(" (detail: ");
111 if (not appendFileErrorDetail(formatter, detail))
112 return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
113 formatter.append(")");
114 }
115 switch (contextKind)
116 {
117 case FileErrorContextKind::None: break;
118 case FileErrorContextKind::NativeError:
119 formatter.append(" (native error: ");
120 formatter.append(static_cast<uint64_t>(context.nativeError));
121 formatter.append(")");
122 break;
123 case FileErrorContextKind::RequiredBytes:
124 formatter.append(" (required capacity: ");
125 formatter.append(static_cast<uint64_t>(context.requiredBytes));
126 formatter.append(" bytes)");
127 break;
128 case FileErrorContextKind::ActualBytes:
129 formatter.append(" (actual bytes: ");
130 formatter.append(static_cast<uint64_t>(context.actualBytes));
131 formatter.append(")");
132 break;
133 default: return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
134 }
135 return formatter.finish();
136}

144inline ResultErrorFormat formatFileError(Result result, Span<char> output)
145{
146 if (result)
147 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
148 if (result.category() != FileResultCategory)
149 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
150 return formatFileError(static_cast<FileError>(result.errorValue()), output);
151}

153inline ResultErrorFormat formatFileError(ResultFile result, Span<char> output)
154{
155 if (result)
156 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
157 if (result.result.category() != FileResultCategory)
158 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
159 return detail::formatFileErrorWithContext(static_cast<FileError>(result.result.errorValue()), result.detail,
160 result.contextKind, result.context, output);
161}
