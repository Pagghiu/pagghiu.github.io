101inline ResultErrorFormat formatAsyncStreamsError(Result result, Span<char> output)
102{
103 if (result)
104 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
105 if (result.category() != AsyncStreamsResultCategory)
106 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
107 return formatAsyncStreamsError(static_cast<AsyncStreamsError>(result.errorValue()), output);
108}
