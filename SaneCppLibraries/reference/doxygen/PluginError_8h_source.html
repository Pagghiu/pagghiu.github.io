149struct [[nodiscard]] ResultPlugin
150{
151 Result result;
152 PluginErrorDetail detail = PluginErrorDetail::None;
153 PluginErrorContextKind contextKind = PluginErrorContextKind::None;
154 PluginErrorContext context = {};
155
156 explicit constexpr ResultPlugin(bool valid = true) : result(valid) {}
157 constexpr ResultPlugin(PluginError error, PluginErrorDetail detail = PluginErrorDetail::None)
158 : result(Result::Error(PluginResultCategory, error)), detail(detail)
159 {}
160 constexpr ResultPlugin(PluginError error, PluginErrorDetail detail, PluginErrorContextKind contextKind,
161 PluginErrorContext context)
162 : result(Result::Error(PluginResultCategory, error)), detail(detail), contextKind(contextKind), context(context)
163 {}
164 constexpr ResultPlugin(Result result) : result(result) {}
165
166 template <typename ResultLike>
167 constexpr ResultPlugin(const ResultLike& other) : result(other.toResult())
168 {}
169
170 static constexpr ResultPlugin withNativeError(PluginError error, PluginErrorDetail detail, uint32_t nativeError)
171 {
172 return {error, detail, PluginErrorContextKind::NativeError, PluginErrorContext(nativeError)};
173 }
174
175 static constexpr ResultPlugin withExitCode(PluginError error, PluginErrorDetail detail, int32_t exitCode)
176 {
177 return {error, detail, PluginErrorContextKind::ExitCode,
178 PluginErrorContext(PluginErrorContext::ExitCode{}, exitCode)};
179 }
180
181 static constexpr ResultPlugin withRequiredBytes(PluginError error, PluginErrorDetail detail, uint32_t requiredBytes)
182 {
183 return {error, detail, PluginErrorContextKind::RequiredBytes,
184 PluginErrorContext(PluginErrorContext::RequiredBytes{}, requiredBytes)};
185 }
186
187 static constexpr ResultPlugin withRequiredElements(PluginError error, PluginErrorDetail detail,
188 uint32_t requiredElements)
189 {
190 return {error, detail, PluginErrorContextKind::RequiredElements,
191 PluginErrorContext(PluginErrorContext::RequiredElements{}, requiredElements)};
192 }
193
194 explicit constexpr operator bool() const { return static_cast<bool>(result); }
195 constexpr operator Result() const { return result; }
196 constexpr Result toResult() const { return result; }
197 constexpr bool isError(PluginError error) const { return result.isError(PluginResultCategory, error); }
198};
