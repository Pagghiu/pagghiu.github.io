322inline ResultErrorFormat formatHttpError(Result result, Span<char> output)
323{
324 if (result)
325 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
326 if (result.category() != HttpResultCategory)
327 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
328 return formatHttpError(static_cast<HttpError>(result.errorValue()), output);
329}
