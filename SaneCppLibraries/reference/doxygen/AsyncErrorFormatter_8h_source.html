99inline ResultErrorFormat formatAsyncError(Result result, Span<char> output)
100{
101 if (result)
102 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
103 if (result.category() != AsyncResultCategory)
104 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
105 return formatAsyncError(static_cast<AsyncError>(result.errorValue()), output);
106}
