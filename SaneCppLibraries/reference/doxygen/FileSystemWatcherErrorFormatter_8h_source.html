67inline ResultErrorFormat formatFileSystemWatcherErrorWithDetails(FileSystemWatcherError error,
68 FileSystemWatcherErrorDetail detail,
69 uint32_t nativeError, Span<char> output)
70{
71 ResultErrorFormatter formatter(output);
72 switch (error)
73 {
74 case FileSystemWatcherError::NotInitialized: formatter.append("Watcher is not initialized"); break;
75 case FileSystemWatcherError::AlreadyWatching: formatter.append("Folder watcher is already watching"); break;
76 case FileSystemWatcherError::NotWatching: formatter.append("Folder watcher is not watching"); break;
77 case FileSystemWatcherError::UnsupportedPathEncoding: formatter.append("Path encoding is not supported"); break;
78 case FileSystemWatcherError::PathPreparationFailed: formatter.append("Failed to prepare path"); break;
79 case FileSystemWatcherError::WatchLimitExceeded: formatter.append("Watch capacity exceeded"); break;
80 case FileSystemWatcherError::BufferTooSmall: formatter.append("Buffer capacity is insufficient"); break;
81 case FileSystemWatcherError::InitializationFailed: formatter.append("Failed to initialize watcher"); break;
82 case FileSystemWatcherError::WatchSetupFailed: formatter.append("Failed to configure watching"); break;
83 case FileSystemWatcherError::StopWatchingFailed: formatter.append("Failed to stop watching"); break;
84 case FileSystemWatcherError::CloseFailed: formatter.append("Failed to close watcher"); break;
85 default: return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
86 }
87
88 if (detail != FileSystemWatcherErrorDetail::None)
89 {
90 formatter.append(" (detail: ");
91 if (not appendFileSystemWatcherErrorDetail(formatter, detail))
92 return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
93 if (nativeError != 0)
94 {
95 formatter.append(", native error: ");
96 formatter.append(static_cast<uint64_t>(nativeError));
97 }
98 formatter.append(")");
99 }
100 else if (nativeError != 0)
101 {
102 formatter.append(" (native error: ");
103 formatter.append(static_cast<uint64_t>(nativeError));
104 formatter.append(")");
105 }
106 return formatter.finish();
107}
108} // namespace detail

110inline ResultErrorFormat formatFileSystemWatcherError(FileSystemWatcherError error, Span<char> output)
111{
112 return detail::formatFileSystemWatcherErrorWithDetails(error, FileSystemWatcherErrorDetail::None, 0, output);

115inline ResultErrorFormat formatFileSystemWatcherError(Result result, Span<char> output)
116{
117 if (result)
118 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
119 if (result.category() != FileSystemWatcherResultCategory)
120 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
121 return formatFileSystemWatcherError(static_cast<FileSystemWatcherError>(result.errorValue()), output);
122}

124inline ResultErrorFormat formatFileSystemWatcherError(ResultFileSystemWatcher result, Span<char> output)
125{
126 if (result)
127 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
128 if (result.result.category() != FileSystemWatcherResultCategory)
129 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
130 return detail::formatFileSystemWatcherErrorWithDetails(
131 static_cast<FileSystemWatcherError>(result.result.errorValue()), result.detail, result.nativeError, output);
132}
