232inline void appendSignedCryptographyContext(ResultErrorFormatter& formatter, int32_t value)
233{
234 if (value < 0)
235 {
236 formatter.append("-");
237 formatter.append(static_cast<uint64_t>(-static_cast<int64_t>(value)));
238 }
239 else
240 formatter.append(static_cast<uint64_t>(value));
241}

243inline ResultErrorFormat formatCryptographyErrorWithContext(CryptographyError error, CryptographyErrorDetail detail,
244 CryptographyErrorContextKind contextKind,
245 CryptographyErrorContext context, Span<char> output)
246{
247 ResultErrorFormatter formatter(output);
248 if (not appendCryptographyError(formatter, error))
249 return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
250 if (detail != CryptographyErrorDetail::None)
251 {
252 formatter.append(" (detail: ");
253 if (not appendCryptographyErrorDetail(formatter, detail))
254 return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
255 formatter.append(")");
256 }
257 switch (contextKind)
258 {
259 case CryptographyErrorContextKind::None: break;
260 case CryptographyErrorContextKind::AppleCommonCryptoStatus:
261 formatter.append(" (CommonCrypto status: ");
262 appendSignedCryptographyContext(formatter, context.appleCommonCryptoStatus);
263 formatter.append(")");
264 break;
265 case CryptographyErrorContextKind::WindowsNtStatus:
266 formatter.append(" (NTSTATUS: ");
267 formatter.append(static_cast<uint64_t>(context.windowsNtStatus));
268 formatter.append(")");
269 break;
270 case CryptographyErrorContextKind::PosixErrno:
271 formatter.append(" (POSIX errno: ");
272 appendSignedCryptographyContext(formatter, context.posixErrno);
273 formatter.append(")");
274 break;
275 case CryptographyErrorContextKind::ExpectedBytes:
276 formatter.append(" (expected: ");
277 formatter.append(static_cast<uint64_t>(context.expectedBytes));
278 formatter.append(" bytes)");
279 break;
280 case CryptographyErrorContextKind::RequiredBytes:
281 formatter.append(" (required capacity: ");
282 formatter.append(static_cast<uint64_t>(context.requiredBytes));
283 formatter.append(" bytes)");
284 break;
285 case CryptographyErrorContextKind::MaximumBytes:
286 formatter.append(" (maximum: ");
287 formatter.append(static_cast<uint64_t>(context.maximumBytes));
288 formatter.append(" bytes)");
289 break;
290 case CryptographyErrorContextKind::ActualBytes:
291 formatter.append(" (actual: ");
292 formatter.append(static_cast<uint64_t>(context.actualBytes));
293 formatter.append(" bytes)");
294 break;
295 default: return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
296 }
297 return formatter.finish();
298}

307inline ResultErrorFormat formatCryptographyError(Result result, Span<char> output)
308{
309 if (result)
310 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
311 if (result.category() != CryptographyResultCategory)
312 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
313 return formatCryptographyError(static_cast<CryptographyError>(result.errorValue()), output);
314}

316inline ResultErrorFormat formatCryptographyError(ResultCryptography result, Span<char> output)
317{
318 if (result)
319 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
320 if (result.result.category() != CryptographyResultCategory)
321 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
322 return detail::formatCryptographyErrorWithContext(static_cast<CryptographyError>(result.result.errorValue()),
323 result.detail, result.contextKind, result.context, output);
324}
