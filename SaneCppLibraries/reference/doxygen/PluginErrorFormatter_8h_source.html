125inline ResultErrorFormat formatPluginErrorWithContext(PluginError error, PluginErrorDetail detail,
126 PluginErrorContextKind contextKind, PluginErrorContext context,
127 Span<char> output)
128{
129 ResultErrorFormatter formatter(output);
130 if (not appendPluginError(formatter, error))
131 return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
132 if (detail != PluginErrorDetail::None)
133 {
134 formatter.append(" (detail: ");
135 if (not appendPluginErrorDetail(formatter, detail))
136 return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
137 formatter.append(")");
138 }
139 switch (contextKind)
140 {
141 case PluginErrorContextKind::None: break;
142 case PluginErrorContextKind::NativeError:
143 formatter.append(" (native error: ");
144 formatter.append(static_cast<uint64_t>(context.nativeError));
145 formatter.append(")");
146 break;
147 case PluginErrorContextKind::ExitCode:
148 formatter.append(" (exit code: ");
149 if (context.exitCode < 0)
150 {
151 formatter.append("-");
152 formatter.append(static_cast<uint64_t>(-static_cast<int64_t>(context.exitCode)));
153 }
154 else
155 formatter.append(static_cast<uint64_t>(context.exitCode));
156 formatter.append(")");
157 break;
158 case PluginErrorContextKind::RequiredBytes:
159 formatter.append(" (required capacity: ");
160 formatter.append(static_cast<uint64_t>(context.requiredBytes));
161 formatter.append(" bytes)");
162 break;
163 case PluginErrorContextKind::RequiredElements:
164 formatter.append(" (required capacity: ");
165 formatter.append(static_cast<uint64_t>(context.requiredElements));
166 formatter.append(" elements)");
167 break;
168 default: return ResultErrorFormatter::failure(ResultErrorFormatStatus::UnknownError, output);
169 }
170 return formatter.finish();
171}

180inline ResultErrorFormat formatPluginError(Result result, Span<char> output)
181{
182 if (result)
183 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
184 if (result.category() != PluginResultCategory)
185 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
186 return formatPluginError(static_cast<PluginError>(result.errorValue()), output);
187}

189inline ResultErrorFormat formatPluginError(ResultPlugin result, Span<char> output)
190{
191 if (result)
192 return ResultErrorFormatter::failure(ResultErrorFormatStatus::NotAnError, output);
193 if (result.result.category() != PluginResultCategory)
194 return ResultErrorFormatter::failure(ResultErrorFormatStatus::ForeignCategory, output);
195 return detail::formatPluginErrorWithContext(static_cast<PluginError>(result.result.errorValue()), result.detail,
196 result.contextKind, result.context, output);
197}
